\section*{Math Notes: Sieve Constraints}

\subsection*{Initial candidate set}

\[
S_0 = \{2,3,4,\dots,N\}.
\]

\subsection*{Layered sieve update}

Let $q_k$ denote the $k$-th prime filter.
\[
S_k = S_{k-1} \setminus \{n : q_k \mid n,\ n \ne q_k\}.
\]

\subsection*{Final retained set}

If all prime filters $q_k \le \sqrt{N}$ are applied, then
\[
S_{\mathrm{final}} = \{p : p \le N,\ p \text{ prime}\}.
\]

\subsection*{Product retention baseline}

\[
M(Q)=\prod_{q \le Q}\left(1-\frac{1}{q}\right).
\]

\subsection*{Log-product decay}

\[
-\log M(Q) = -\log \prod_{q \le Q}\left(1-\frac{1}{q}\right).
\]

\subsection*{Log-log slope fit}
\[
-\log M(Q) \approx a\log\log Q + b.
\]
Measured fit: $a = 0.815598$ and $R^2 = 0.992105$.
\subsection*{Slope versus max filter prime}
\[
a(Q_{\max}) \text{ is fitted from } -\log M(Q) \approx a(Q_{\max})\log\log Q + b.
\]
Final prefix slope: $a = 0.815598$.
\subsection*{Product-baseline error}

\[
error(Q) = retention_{\mathrm{observed}}(Q) - M(Q).
\]

\subsection*{Retention and drift}

\[
retention_k = \frac{|S_k|}{|S_0|}.
\]
\[
drift_k = 1 - retention_k.
\]

\subsection*{CGCS score}

Let $P_N = \{p : p \le N\}$.
\[
CGCS_{\mathrm{sieve}} =
\frac{|S_{\mathrm{final}} \cap P_N|}{|P_N|}.
\]

Measured value:
\[
CGCS_{\mathrm{sieve}} = 1.000000.
\]

\subsection*{Recoverability}

The full sieve recovers prime identity exactly up to $N$, provided all filters $q \le \sqrt{N}$ are applied.
