\section*{Math Notes: Constraint Reconstruction}

\subsection*{Prime set}
\[P_N = \{p : p \le N\}.\]

\subsection*{Candidate reconstruction}
\[\hat{P}_N = F(S_{\mathrm{raw}}).\]

\subsection*{Precision and recovery}
\[precision = \frac{|\hat{P}_N \cap P_N|}{|\hat{P}_N|}.\]
\[recovery = \frac{|\hat{P}_N \cap P_N|}{|P_N|}.\]

\subsection*{$F_1$ score}
\[F_1 = \frac{2\cdot precision\cdot recovery}{precision+recovery}.\]

\subsection*{Residue filter}
\[n=2,3 \quad \text{or} \quad n\equiv 1,5\pmod6.\]

\subsection*{Density-guided completion}
\[\pi_{\mathrm{model}}(x)\approx \frac{x}{\log x - 1}.\]

\subsection*{Gap diagnostic}
\[drift_{\mathrm{gap}}=\operatorname{mean}_i \frac{|g_i-\log p_i|}{\log p_i}.\]

Candidate completions are hypotheses, not proofs of primality.
